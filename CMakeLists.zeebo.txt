# CMakeLists.zeebo.txt - Zeebo / BREW 4.0.2 build configuration for GTA III (re3).
#
# Usage:
#   cmake -S . -B build/zeebo -G Ninja -DCMAKE_TOOLCHAIN_FILE=cmake/zeebo-arm.cmake -DCMAKE_BUILD_TYPE=Release
#   (invoke with: cmake -S . -B build/zeebo -G Ninja -DCMAKE_TOOLCHAIN_FILE=cmake/zeebo-arm.cmake)
#   or rename/pass -C if desired.
cmake_minimum_required(VERSION 3.16)
project(gta3_zeebo C CXX ASM)

set(CMAKE_CXX_STANDARD_REQUIRED ON)

    ${CMAKE_CURRENT_SOURCE_DIR}/vendor/librw
    ${CMAKE_CURRENT_SOURCE_DIR}/src/platform/zeebo
)

    __ZEEBO__=1
    RW_GLES1=1
)

target_compile_options(librw_zeebo PRIVATE

    -w
    -fpermissive
    -fsigned-char

    -fno-threadsafe-statics

    -fdata-sections

# ---------------------------------------------------------------------------
# 2. re3 sources (excluding Win32 / GLFW / POSIX pthreads / OpenAL / MSS)
# ---------------------------------------------------------------------------
file(GLOB_RECURSE RE3_ALL_SOURCES
    "src/*.cpp"
    "src/*.c"
)

set(RE3_ZEEBO_SOURCES "")
foreach(SRC ${RE3_ALL_SOURCES})
    file(RELATIVE_PATH REL "${CMAKE_CURRENT_SOURCE_DIR}" "${SRC}")
    if(REL MATCHES "^src/skel/win/"
       OR REL MATCHES "^src/skel/glfw/"
       OR REL STREQUAL "src/core/CdStream.cpp"
       OR REL STREQUAL "src/core/CdStreamPosix.cpp"
       OR REL STREQUAL "src/audio/sampman_miles.cpp"
       OR REL STREQUAL "src/audio/sampman_oal.cpp"
       OR REL STREQUAL "src/audio/sampman_null.cpp"
       OR REL MATCHES "^src/audio/oal/"
       OR REL MATCHES "^src/audio/eax/"
       OR REL STREQUAL "src/extras/custompipes_d3d9.cpp"
       OR REL STREQUAL "src/extras/custompipes_gl.cpp"
       OR REL STREQUAL "src/extras/GitSHA1.cpp")
        continue()
    endif()
    list(APPEND RE3_ZEEBO_SOURCES "${SRC}")
endforeach()

set(GIT_SHA1 "zeebo-port")
configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/src/extras/GitSHA1.cpp.in"
    "${CMAKE_CURRENT_BINARY_DIR}/extras/GitSHA1.cpp"
    @ONLY
)
list(APPEND RE3_ZEEBO_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/extras/GitSHA1.cpp")

# Collect all src subdirectories for re3's flat #include style
file(GLOB_RECURSE RE3_HEADERS "src/*.h")
set(RE3_INCLUDES "")
foreach(HDR ${RE3_HEADERS})
    file(RELATIVE_PATH REL_H "${CMAKE_CURRENT_SOURCE_DIR}" "${HDR}")
    if(REL_H MATCHES "^src/skel/win/"
       OR REL_H MATCHES "^src/skel/glfw/"
       OR REL_H MATCHES "^src/audio/oal/"
       OR REL_H MATCHES "^src/audio/eax/")
        continue()
    endif()
    get_filename_component(DIR "${HDR}" DIRECTORY)
    list(APPEND RE3_INCLUDES "${DIR}")
endforeach()
list(REMOVE_DUPLICATES RE3_INCLUDES)

add_executable(gta3zeebo.elf ${RE3_ZEEBO_SOURCES})

target_link_libraries(gta3zeebo.elf PRIVATE librw_zeebo)

target_include_directories(gta3zeebo.elf PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/vendor/pl_mpeg
    ${RE3_INCLUDES}
)

target_compile_definitions(gta3zeebo.elf PRIVATE
    __ZEEBO__=1
    RW_GLES1=1
    LIBRW=1
    NDEBUG=1
    CMAKE_BUILD=1
    USE_OUR_VERSIONING=1
    CMAKE_NO_AUTOLINK=1
)

target_compile_options(gta3zeebo.elf PRIVATE
    -O2
    -w
    -fpermissive
    -fsigned-char
    -fno-strict-aliasing
    -fno-strict-overflow
    -fno-exceptions
    -fno-rtti
    -fno-threadsafe-statics
    -fsingle-precision-constant
    -fno-math-errno
    -fno-trapping-math
    -ffunction-sections
    -fdata-sections
)

target_link_options(gta3zeebo.elf PRIVATE
    -T${CMAKE_CURRENT_SOURCE_DIR}/tools/zeebo/zeebo_mod.ld
    -nostartfiles
    -Wl,--gc-sections
)

if(ZEEBO_OBJCOPY)
    add_custom_command(TARGET gta3zeebo.elf POST_BUILD
        COMMAND ${ZEEBO_OBJCOPY} -O binary $<TARGET_FILE:gta3zeebo.elf> ${CMAKE_CURRENT_BINARY_DIR}/gta3zeebo.mod
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_BINARY_DIR}/gta3zeebo.mod ${CMAKE_CURRENT_SOURCE_DIR}/dist/zeebo/gta3zeebo/gta3zeebo.mod
        COMMENT "Generating BREW flat binary gta3zeebo.mod and syncing to dist/zeebo/gta3zeebo/"
    )
endif()
